// Firebase Console → Firestore → Rules → Publish (bắt buộc sau khi bật Authentication)

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /cms/{document} {
      // Khách xem web: chỉ đọc
      allow read: if true;

      // Chỉ tài khoản đã đăng nhập Firebase mới được ghi (admin)
      allow write: if request.auth != null;
    }
  }
}
